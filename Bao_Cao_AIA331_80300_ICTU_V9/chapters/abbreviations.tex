\chapter*{DANH MỤC TỪ VIẾT TẮT}
\addcontentsline{toc}{chapter}{DANH MỤC TỪ VIẾT TẮT}

\begin{table}[h!]
\centering
\small
\caption{Các từ viết tắt sử dụng trong tài liệu}\label{tab:v8-abbreviations}
\begin{tabularx}{\textwidth}{|L{2.8cm}|Y|}
\hline
\thead{Từ viết tắt} & \thead{Ý nghĩa}\\
\hline
AI & Artificial Intelligence -- trí tuệ nhân tạo.\\
\hline
API & Application Programming Interface -- giao diện lập trình ứng dụng.\\
\hline
CI/CD & Continuous Integration / Continuous Deployment -- quy trình tích hợp và phát hành tự động.\\
\hline
CORS & Cross-Origin Resource Sharing -- cơ chế chia sẻ tài nguyên giữa các nguồn khác nhau.\\
\hline
CRUD & Create, Read, Update, Delete -- bốn thao tác dữ liệu cơ bản.\\
\hline
C4 & Context, Container, Component, Code -- mô hình mô tả kiến trúc phần mềm phân tầng.\\
\hline
ERD & Entity--Relationship Diagram -- sơ đồ thực thể--liên kết quan hệ dữ liệu.\\
\hline
HITL & Human-in-the-Loop -- cơ chế con người tham gia thẩm định và phê duyệt trong quy trình AI.\\
\hline
IDOR & Insecure Direct Object References -- lỗ hổng bảo mật tham chiếu đối tượng trực tiếp.\\
\hline
JWT & JSON Web Token -- chuẩn mở định dạng chuỗi xác thực và ủy quyền người dùng an toàn.\\
\hline
KPI & Key Performance Indicator -- chỉ số hiệu quả then chốt trong kinh doanh và marketing.\\
\hline
LLM & Large Language Model -- mô hình ngôn ngữ lớn.\\
\hline
NFR & Non-Functional Requirement -- yêu cầu phi chức năng (bảo mật, hiệu năng, độ tin cậy).\\
\hline
ORM & Object-Relational Mapping -- kỹ thuật ánh xạ đối tượng lập trình với cơ sở dữ liệu quan hệ.\\
\hline
RBAC & Role-Based Access Control -- kiểm soát truy cập và phân quyền theo vai trò người dùng.\\
\hline
REST & Representational State Transfer -- phong cách kiến trúc dịch vụ web chuẩn hóa giao thức HTTP.\\
\hline
SDLC & Software Development Life Cycle -- vòng đời phát triển phần mềm chuẩn mực.\\
\hline
SPA & Single-Page Application -- kiến trúc ứng dụng web một trang tải động giao diện mượt mà.\\
\hline
WAL & Write-Ahead Logging -- chế độ ghi nhật ký trước của CSDL giúp tối ưu thông lượng đọc/ghi.\\
\hline
\end{tabularx}
\end{table}
