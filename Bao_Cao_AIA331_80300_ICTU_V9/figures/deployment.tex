\begin{tikzpicture}[

% Cột 1: Client Node (x = 1.9cm, y = 0)
\node[node_box] (client) at (1.9, 0) {

  \fbox{\tiny FastAPI Engine / Python}

\node[node_box] (database) at (7.0, -1.5) {
