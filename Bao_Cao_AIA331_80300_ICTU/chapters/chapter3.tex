\section{Danh mục yêu cầu chất lượng (NFR)}

Hệ thống tuân thủ 6 yêu cầu phi chức năng (NFR01--NFR06) đã được kiểm chứng thực nghiệm:
\begin{itemize}
 \item \textbf{NFR01 (Kiểm soát truy cập đa tầng và chống xung đột lợi ích)}: Kết hợp RBAC 2 vai trò (\texttt{MANAGER}, \texttt{MARKETER}), kiểm soát quyền sở hữu mức bản ghi (\texttt{owner\_id}) và quy tắc chống tự duyệt (\texttt{CANNOT\_REVIEW\_OWN}).
 \item \textbf{NFR02 (Toàn vẹn dữ liệu và giao dịch)}: Ràng buộc khóa ngoại, kiểm tra bất biến số học phễu tiếp thị (\texttt{views >= clicks >= conversions >= 0}) và khóa xóa nội dung đã duyệt (\texttt{CONTENT\_LOCKED}).
 \item \textbf{NFR03 (Hiệu năng và giới hạn tài nguyên)}: Các điểm cuối REST API nghiệp vụ phản hồi trung bình dưới 30ms; dịch vụ \texttt{ai-runner} giới hạn hàng đợi tuần tự \texttt{RUNNER\_CONCURRENCY=1} và trần bộ nhớ \texttt{MemoryMax=896M} dựa trên kết quả đo P0 Spike (\texttt{\~{}490 MB RSS}).
 \item \textbf{NFR04 (Trải nghiệm giao diện và truyền phát SSE)}: Giao diện tiếng Việt chuẩn hóa, truyền phát sự kiện hội thoại AI theo thời gian thực qua SSE, hiển thị trực quan thẻ công cụ và thẻ đề xuất.
 \item \textbf{NFR05 (Khả năng chịu lỗi và đa nhà cung cấp AI)}: Hỗ trợ 13 mô hình LLM trên 5 nền tảng với cơ chế tự động chuyển đổi dự phòng (failover); khi \texttt{ai-runner} ngoại tuyến, hệ thống báo lỗi chuẩn hóa mà không ảnh hưởng đến toàn bộ chức năng CRUD.
 \item \textbf{NFR06 (Khả năng tái lập và kiểm thử tự động)}: Đóng gói Docker Compose, kịch bản tạo dữ liệu mẫu \texttt{seed\_demo.py} và 151 ca kiểm thử tự động (Pytest + Playwright E2E) vượt qua 100\%.
\end{itemize}

\subsection{Cơ chế xác thực kép: User JWT và Ephemeral Run-Token JWT}
Trong mô-đun \texttt{backend/app/core/security.py} và \texttt{backend/app/api/deps.py}, hệ thống vận hành hai loại mã thông báo JWT tách biệt:
\begin{enumerate}
 \item \textbf{Access Token cho người dùng (\texttt{token\_version})}: Khi người dùng gọi API, hàm \texttt{get\_current\_user} giải mã token, truy vấn bảng \texttt{users} và xác nhận \texttt{user.is\_active == True} cùng \texttt{token\_version} khớp với CSDL. Nếu tài khoản bị khóa hoặc đã đăng xuất, yêu cầu bị từ chối ngay lập tức.
 \item \textbf{Run-Token tạm thời cho tiến trình AI (\texttt{typ="assistant\_run"})}: Khi khởi chạy lượt hội thoại, hàm \texttt{create\_run\_token} sinh một JWT có hiệu lực đúng 10 phút mang theo \texttt{uid=user.id} và \texttt{thread\_id}. Máy chủ FastMCP tại \texttt{/mcp} giải mã token này trong mỗi lần gọi công cụ để áp dụng đúng phạm vi quyền của người dùng đang chat.
\end{enumerate}

\subsection{Quy tắc kiểm soát quyền sở hữu (\texttt{owner\_id}) và chống tự duyệt (\texttt{CANNOT\_REVIEW\_OWN})}
\begin{enumerate}
 \item \textbf{Quyền sở hữu Chiến dịch (\texttt{owner\_id})}: Trong \texttt{backend/app/services/campaign\_service.py}, \texttt{MARKETER} chỉ được phép chỉnh sửa, xóa, thêm nội dung, lập lịch hoặc nhập số liệu trên chiến dịch mà \texttt{campaign.owner\_id == current\_user.id}. Mọi nỗ lực can thiệp vào chiến dịch của người khác đều bị từ chối với mã lỗi HTTP 403 (\texttt{PERMISSION\_DENIED}).
 \item \textbf{Quy tắc chống tự duyệt (\texttt{CANNOT\_REVIEW\_OWN})}: Trong \texttt{backend/app/services/content\_service.py}, ngay cả khi người dùng có vai trò \texttt{MANAGER}, nếu họ là người tạo ra bài viết (\texttt{content.created\_by == current\_user.id}), hệ thống sẽ chặn thao tác \texttt{POST /contents/\{id\}/approve} hoặc \texttt{/reject} với mã lỗi \texttt{CANNOT\_REVIEW\_OWN}. Quy tắc này đảm bảo nguyên tắc ``bốn mắt'' (\textit{four-eyes principle}) trong kiểm duyệt truyền thông.
\end{enumerate}

\subsection{Các trạng thái và thuộc tính nguồn gốc}
\begin{itemize}
 \item \texttt{DRAFT}: Bản nháp mới khởi tạo (có thuộc tính \texttt{origin = MANUAL} nếu soạn tay, hoặc \texttt{origin = AI} kèm \texttt{ai\_message\_id} nếu được xác nhận từ thẻ đề xuất của Trợ lý AI).
 \item \texttt{PENDING\_REVIEW}: Bài viết đã được tác giả gửi lên hàng đợi phê duyệt qua \texttt{POST /contents/\{id\}/submit}.
 \item \texttt{APPROVED}: Bài viết đã được một \texttt{MANAGER} khác thẩm định và phê duyệt qua \texttt{POST /contents/\{id\}/approve}.
 \item \texttt{REJECTED}: Bài viết bị từ chối qua \texttt{POST /contents/\{id\}/reject} (bắt buộc kèm lý do phản hồi trong \texttt{content\_reviews.comment}).
 \item \texttt{PUBLISHED}: Bài viết đã được xác nhận xuất bản qua \texttt{POST /contents/\{id\}/mark-published} kèm đường dẫn \texttt{published\_url}.
\end{itemize}

\subsection{Các rào chắn toàn vẹn quy trình (Workflow Guardrails)}
\begin{enumerate}
 \item \textbf{Tự động quét từ khóa cấm khi gửi duyệt}: Khi gọi \texttt{POST /contents/\{id\}/submit}, hệ thống tự động đối chiếu tiêu đề, nội dung và CTA với bảng \texttt{banned\_keywords}. Nếu phát hiện từ khóa vi phạm, hệ thống chuyển trạng thái sang \texttt{PENDING\_REVIEW} và trả kèm danh sách \texttt{banned\_matches} cảnh báo cụ thể để Người quản lý xem xét.
 \item \textbf{Tự động thu hồi phê duyệt khi sửa nội dung đã duyệt}: Nếu một bài viết đang ở trạng thái \texttt{APPROVED} hoặc \texttt{REJECTED} bị chỉnh sửa nội dung, hệ thống tự động đưa trạng thái về lại \texttt{DRAFT}, buộc phải gửi duyệt lại từ đầu.
 \item \textbf{Khóa xóa đối với nội dung đã duyệt hoặc đã xuất bản (\texttt{CONTENT\_LOCKED})}: Điểm cuối \texttt{DELETE /contents/\{id\}} từ chối xóa mọi bản ghi có trạng thái \texttt{APPROVED} hoặc \texttt{PUBLISHED} với mã lỗi HTTP 409 Conflict (\texttt{CONTENT\_LOCKED}), bảo vệ toàn vẹn lịch sử kiểm toán và dữ liệu xuất bản.
 \item \textbf{Đồng bộ tự động với Lịch xuất bản (\texttt{schedules})}: Khi bài viết chuyển sang \texttt{PUBLISHED} qua \texttt{/mark-published}, các lịch đăng đang ở trạng thái \texttt{PLANNED} gắn với bài viết đó sẽ tự động được cập nhật sang \texttt{DONE}.
\end{enumerate}

\section{Kết luận chương}

Chương 3 đã hoàn thiện phân tích chi tiết toàn bộ các yêu cầu nghiệp vụ và phi chức năng của MarketFlow AI. Sự kết hợp chặt chẽ giữa RBAC 2 vai trò (\texttt{MANAGER}, \texttt{MARKETER}), kiểm soát quyền sở hữu (\texttt{owner\_id}), quy tắc chống tự duyệt (\texttt{CANNOT\_REVIEW\_OWN}) và máy trạng thái duyệt nội dung HITL 5 trạng thái đã thiết lập nên một kiến trúc bảo mật đa tầng vững chắc. Chương tiếp theo sẽ trình bày thiết kế kiến trúc tổng thể, lược đồ CSDL 14 bảng và thiết kế 14 công cụ MCP của hệ thống.
