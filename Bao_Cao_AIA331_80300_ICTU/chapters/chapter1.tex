\chapter{PHÂN TÍCH BÀI TOÁN VÀ YÊU CẦU HỆ THỐNG}

\section{Bài toán quản lý thực tế}

Tại các nhóm marketing nhỏ, một chiến dịch quảng cáo thường chạy song song trên nhiều kênh truyền thông như Facebook, Instagram, TikTok, Zalo OA, Email, Google Ads và Website. Quy trình làm việc hàng ngày gồm 5 bước nối tiếp nhau:
\begin{enumerate}[noitemsep,topsep=2pt]
 \item Tạo chiến dịch gắn với sản phẩm, đối tượng khách hàng, mục tiêu (\texttt{SALES}, \texttt{LEADS}, \texttt{TRAFFIC}, \texttt{AWARENESS}), thời gian chạy và tổng ngân sách.
 \item Chọn các kênh truyền thông tham gia chiến dịch và chia ngân sách dự kiến cho từng kênh.
 \item Soạn bài viết (tiêu đề, nội dung chính, lời kêu gọi hành động CTA, hashtag) theo quy định độ dài của từng kênh và gửi cho quản lý duyệt.
 \item Lên lịch đăng cho các bài viết đã được duyệt và cập nhật đường dẫn bài đăng thực tế sau khi phát hành.
 \item Ghi nhận số liệu quảng cáo theo ngày (\texttt{views}, \texttt{clicks}, \texttt{conversions}, \texttt{cost}) để tính các chỉ số hiệu quả (CTR, CVR, CPC, CPA, CPM) và theo dõi tiến độ chi tiêu ngân sách.
\end{enumerate}

Khi làm thủ công qua nhiều file bảng tính và tin nhắn rời rạc, nhóm làm việc thường gặp 3 vấn đề chính:
\begin{itemize}[noitemsep,topsep=2pt]
 \item \textbf{Lệch dữ liệu và sai số khi nhập báo cáo}: Ngân sách kế hoạch nằm ở một file, kết quả quảng cáo nằm ở các file CSV xuất ra từ từng kênh. Khi nhập tay hoặc gộp file dễ xảy ra lỗi dữ liệu vô lý (ví dụ số lượt nhấp lớn hơn số lượt hiển thị, hoặc số chuyển đổi lớn hơn số lượt nhấp).
 \item \textbf{Thiếu bước kiểm soát từ cấm và phân quyền duyệt bài}: Nhân viên viết bài dễ vô tình dùng các cụm từ quảng cáo quá đà bị cấm (như \textit{``cam kết 100\%''}, \textit{``thần dược''}, \textit{``rẻ nhất''}), hoặc người viết tự đăng bài khi chưa có người khác kiểm tra.
 \item \textbf{AI bên ngoài không nắm được dữ liệu thật của chiến dịch}: Khi dùng các trang chat AI rời rạc, nhân viên phải tự chép mục tiêu và số liệu sang cửa sổ chat. Nếu chiến dịch chưa có số liệu, AI rất dễ tự bịa ra con số giả định hoặc viết bài không đúng giọng văn thương hiệu.
\end{itemize}

Hệ thống \textbf{MarketFlow} (Đề tài 80300) được xây dựng để giải quyết trực tiếp 3 vấn đề trên: quản lý tập trung toàn bộ dữ liệu chiến dịch trên một cơ sở dữ liệu thống nhất, kiểm tra từ cấm và bắt buộc người duyệt bài phải khác người tạo bài, đồng thời tích hợp Trợ lý AI đọc trực tiếp số liệu thật trong hệ thống để gợi ý ý tưởng, viết bài nháp và tóm tắt kết quả chiến dịch.

\subsection{Yêu cầu phi chức năng}

\begin{itemize}[noitemsep,topsep=2pt]
 \item \textbf{NFR01 (Bảo mật và phân quyền)}: Mật khẩu được băm bằng Bcrypt. API kiểm tra quyền theo vai trò (\texttt{MANAGER}/\texttt{MARKETER}) và theo người phụ trách chiến dịch (\texttt{owner\_id}). Khóa API của các nhà cung cấp AI chỉ lưu ở máy chủ, không lộ ra trình duyệt.
 \item \textbf{NFR02 (Kiểm soát tính hợp lệ của dữ liệu)}: Kiểm tra chặt các điều kiện ngày tháng, ngân sách không âm, điều kiện số liệu quảng cáo (\texttt{views >= clicks >= conversions}) ở cả tầng kiểm tra đầu vào (Pydantic) và tầng cơ sở dữ liệu.
 \item \textbf{NFR03 (Hiệu năng và giới hạn bộ nhớ)}: Các API quản lý phản hồi nhanh dưới 50ms. Dịch vụ chạy AI (\texttt{ai-runner}) được giới hạn chạy tuần tự từng yêu cầu (\texttt{RUNNER\_CONCURRENCY=1}) và giới hạn mức RAM tối đa \texttt{896M} để không làm treo máy chủ VPS.
 \item \textbf{NFR04 (Giao diện dễ thao tác)}: Giao diện tiếng Việt thống nhất, có thông báo lỗi rõ ràng ở từng biểu mẫu. Phần chat với Trợ lý AI hiển thị kết quả trực tiếp theo luồng sự kiện thời gian thực (SSE).
 \item \textbf{NFR05 (Tự động đổi mô hình AI khi lỗi)}: Kết nối 13 mô hình AI miễn phí trên 4 nền tảng. Khi một mô hình hết lượt gọi (HTTP 429) hoặc quá thời gian chờ, hệ thống tự chuyển sang mô hình dự phòng tiếp theo, và nếu dịch vụ AI tắt thì các chức năng quản lý vẫn hoạt động bình thường.
 \item \textbf{NFR06 (Dễ cài đặt và chạy thử)}: Có sẵn cấu hình Docker Compose, kịch bản tạo dữ liệu mẫu (\texttt{seed\_demo.py}) và bộ 151 bài kiểm thử tự động.
\end{itemize}
